492 iterator erase(const_iterator pos) {
493 if (pos == end()) {
494 throw std::out_of_range("Cannot erase end iterator");
495 }
496
497 node_ptr node_to_erase = pos.get_node();
498 std::vector<node_ptr> update(MAX_LEVEL, head_);
499 node_ptr current = head_;
500
501 for (int i = max_level_; i >= 0; --i) {
502 while (current->forward[i] && current->forward[i]->data < node_to_erase->data) {
503 current = current->forward[i];
504 }
505 update[i] = current;
506 }
507
508 // Check that node is actually found
509 if (!current->forward[0] || current->forward[0] != node_to_erase) {
510 throw std::runtime_error("Node not found in skip list");
511 }
512
513 for (size_type i = 0; i <= node_to_erase->level; ++i) {
514 update[i]->forward[i] = node_to_erase->forward[i];
515 }
516
517 while (max_level_ > 0 && head_->forward[max_level_] == nullptr) {
518 --max_level_;
519 }
520
521 --size_;
522 return iterator(node_to_erase->forward[0]);
523 }

633 template<typename U>
634 std::pair<iterator, bool> insert_impl(U&& value) {
635 std::vector<node_ptr> update(MAX_LEVEL, head_);
636 node_ptr current = head_;
637
638 for (int i = max_level_; i >= 0; --i) {
639 while (current->forward[i] && current->forward[i]->data < value) {
640 current = current->forward[i];
641 }
642 update[i] = current;
643 }
644
645 current = current->forward[0];
646
647 if (current && current->data == value) {
648 return {iterator(current), false};
649 }
650
651 size_type new_level = random_level();
652 if (new_level > max_level_) {
653 for (size_type i = max_level_ + 1; i <= new_level; ++i) {
654 update[i] = head_;
655 }
656 max_level_ = new_level;
657 }
658
659 auto new_node = std::make_shared<skip_node>(std::forward<U>(value), new_level);
660
661 for (size_type i = 0; i <= new_level; ++i) {
662 new_node->forward[i] = update[i]->forward[i];
663 update[i]->forward[i] = new_node;
664 }
665
666 ++size_;
667 return {iterator(new_node), true};
668 }
